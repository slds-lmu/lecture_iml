\begin{framev}[align=top]{Beyond LIME: Which Part Should We Improve?}
\textbf{Recall the pipeline:}
Features $\rightarrow$ samples $\rightarrow$ surrogate $\rightarrow$ explanation.\\
\spacer[0.25]
The survey by Knab et al.\ organizes follow-up work by \textbf{where it changes this pipeline} and \textbf{which issue it addresses}. \furtherreading{KNAB2025LIMESURVEY}
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.27\textwidth} >{\raggedright\arraybackslash}p{0.40\textwidth} >{\raggedright\arraybackslash}p{0.25\textwidth}@{}}
\toprule
\textbf{Pitfall} & \textbf{Follow-up intuition} & \textbf{Examples} \\
\midrule
Unrealistic samples & Learn what plausible neighbors look like & \textcolor{limeBlue}{\textbf{MeLIME}} \\
Locality; local vs.\ global signals & Sample at the right scale or near the boundary & \textcolor{limeTeal}{\textbf{s-LIME}; LS} \\

\bottomrule
\end{tabular}
\end{center}
\textbf{Focus:} Representative ideas, each targeting a particular weakness.
\end{framev}

\begin{itemizeM}

\item \textbf{How:} For example, use a density estimator or a generative model.
\item \textbf{Limit:} Plausibility depends on the learned distribution and available data; rare regions may be poorly represented.
\end{itemizeM}

\spacer[0]
}
\par
\spacer[0.25]
\textit{Schematic: one feature; vertical jitter separates sampled points.}
\begin{itemizeM}
\item \textbf{Idea (Smoothed LIME):} Draw continuous perturbations from a distribution whose spread sets locality; fit with equal weights. \furtherreading{GAUDEL2022}
\item \textbf{Why:} Zooming in still provides a useful set of local samples.
\item \textbf{Limit:} Local sampling alone does not preserve feature dependencies. For trees, a tiny neighborhood may show no prediction change.
\end{itemizeM}
\textbf{Related idea:} LS, covered in the pitfalls deck, samples near a decision boundary to capture locally relevant features. \furtherreading{LAUGEL2018}
\end{framev}


\begin{framev}[align=top]{Stability: S-LIME and BayLIME}
\textbf{Pitfall:} Repeating the explanation for the \textbf{same input} can select different features.\\
\spacer[0.25]
\textbf{S-LIME (Stabilized LIME):} Collect enough evidence before choosing a feature. \furtherreading{ZHOU2021SLIME}
\begin{center}
\begin{tikzpicture}[every node/.style={align=center},followupflow/.style={draw=limeBlue!65,rounded corners,fill=limeBlue!7,text width=2.85cm,minimum height=1cm,inner sep=4pt}]
\node[followupflow] (sample) {Draw perturbations};
\node[followupflow,right=0.45cm of sample] (compare) {Compare competing\\features statistically};
\node[followupflow,right=0.45cm of compare] (more) {Too close to call?\\Increase sample size};
\draw[->,thick,limeBlue] (sample) -- (compare);
\draw[->,thick,limeBlue] (compare) -- (more);
\draw[->,thick,limeBlue] (more.south) -- ++(0,-0.45) -| (sample.south);
\end{tikzpicture}
\end{center}
\begin{itemizeM}
\item \textbf{BayLIME:} Combine evidence from perturbations with prior knowledge in a Bayesian surrogate, reducing dependence on one random sample. \furtherreading{ZHAO2021BAYLIME}
\item \textbf{Trade-offs:} More samples cost model queries; prior knowledge can help only to the extent that it is appropriate.
\end{itemizeM}
\spacer[0.25]
\textbf{Scope:} Repeatability at one input does not establish accuracy or robustness across nearby inputs.\\
\spacer[0.25]
\textbf{Naming:} \textbf{s-LIME} changes locality; \textbf{S-LIME} stabilizes feature selection.
\end{framev}


\begin{framev}[align=top]{LIMEtree: Allow the Surrogate to Branch}
\textbf{Pitfall:} One sparse linear model may miss interactions and nonlinear behavior.\\
\spacer[0.25]
\splitVTT[0.48]{
\spacer[0]
\textbf{A single linear explanation}\\
\spacer[0.25]
\enquote{Longer loan duration lowers the predicted approval probability.}\\
\spacer[0.25]
The same slope applies throughout the chosen neighborhood.
}{
\spacer[0]
\centering
\textbf{A tree can express different cases}\\
\spacer[0.25]
\begin{tikzpicture}[every node/.style={align=center,font=\footnotesize},treenode/.style={draw=limeTeal!65,rounded corners,fill=limeTeal!7,inner sep=4pt,text width=1.6cm},treevalue/.style={treenode,text width=1cm}]
\node[treenode] (root) at (-0.5,0) {Short duration?};
\node[treevalue] (left) at (-1.65,-1.05) {$0.8$};
\node[treenode] (right) at (1,-1.05) {High income?};
\node[treevalue] (higher) at (0.2,-2.25) {$0.6$};
\node[treevalue] (lower) at (1.9,-2.25) {$0.2$};
\draw[->,thick,limeTeal] (root) -- node[left,font=\footnotesize] {yes} (left);
\draw[->,thick,limeTeal] (root) -- node[right,font=\footnotesize] {no} (right);
\draw[->,thick,limeTeal] (right) -- node[left] {yes} (higher);
\draw[->,thick,limeTeal] (right) -- node[right] {no} (lower);
\end{tikzpicture}
\spacer[0]
}
\par
\spacer[0.25]
\textit{Toy example: leaves give predicted approval probabilities.}
\begin{itemizeM}
\item \textbf{Idea:} LIMEtree uses a local multi-output regression tree: one tree approximates probabilities for multiple classes. \furtherreading{SOKOL2025LIMETREE}
\item \textbf{Why:} Branches can capture interactions and provide rules or what-if explanations from a shared surrogate.
\item \textbf{Limit:} A richer surrogate can improve fit, but a larger tree is harder to interpret. High fidelity still needs to be checked in the relevant neighborhood.
\end{itemizeM}
\end{framev}

\par
\spacer[0.25]
\textit{Schematic: patch-based segmentation versus an object/part hierarchy.}
\begin{itemizeM}
\item \textbf{Idea:} Use a segmentation model, such as Segment Anything, to obtain regions; organize them hierarchically and refine important regions. \furtherreading{KNAB2024DSEGLIME}
\item \textbf{Why:} Users can inspect a coarse explanation, then explore relevant parts.
\item \textbf{Limit:} The segmenter and masking baseline still influence the result. Meaningful regions do not by themselves ensure faithful explanations.
\end{itemizeM}
\end{framev}


\begin{framev}[align=top]{Which LIME Should I Trust?}
\textbf{Start with the pitfall you need to address.}
\begin{itemizeM}
\item \textbf{Better samples and locality:} MeLIME learns plausible neighbors; s-LIME generates them at the requested scale. These address different problems.
\item \textbf{Better estimation and representation:} S-LIME and BayLIME target instability; LIMEtree changes the surrogate; DSEG-LIME changes image features.
\item \textbf{Check fit and variation separately:} Test agreement on fresh local samples; repeat explanations and inspect nearby inputs with similar predictions.
\end{itemizeM}
\spacer[0.25]
\textbf{Hiding biases remains a concern:}
Data-informed sampling addresses the unrealistic perturbations exploited by the OOD attack. It does not certify fairness or resistance to manipulation. \furtherreading{SLACK2020}\\
\spacer[0.25]
\textbf{Further reading:} \emph{Which LIME should I trust?} (Knab et al., 2025) maps variants across \textbf{locality, fidelity, interpretability, stability, and efficiency}, and highlights gaps in comparative evaluation. \furtherreading{KNAB2025LIMESURVEY}\\
\spacer[0.25]
\textbf{Explore the overview:}
\sourceref{https://patrick-knab.github.io/which-lime-to-trust/}
\end{framev}
